def inring(x,y,ps):
 inside=False
 for (ax,ay),(bx,by) in zip(ps,ps[1:]+ps[:1]):
  # Boundary is a separate result; test witnesses never use a boundary.
  cr=(bx-ax)*(y-ay)-(by-ay)*(x-ax)
  if cr==0 and min(ax,bx)<=x<=max(ax,bx) and min(ay,by)<=y<=max(ay,by):return None
  if (ay>y)!=(by>y):
   xx=F(ax)+F(bx-ax)*(y-ay)/F(by-ay)
   if xx>x:inside=not inside
 return inside
def inside(x,y,rs):

  hit=inring(x,y,row['outline'])

  if hit:
   hole=False

    v=inring(x,y,h)
    if v is None:return None
    hole=hole or v
   if not hole:return True
 return False

def nodes(s):
